const CHUNK: u64 = 65_536;
fn mix(mut n: u64) -> u64 {
    n = n.wrapping_add(0x9e3779b97f4a7c15);
    n = (n ^ (n >> 30)).wrapping_mul(0xbf58476d1ce4e5b9);
    n = (n ^ (n >> 27)).wrapping_mul(0x94d049bb133111eb);
    n ^ (n >> 31)
}
fn chunk(first: u64, rows: u64, multi: bool, dense: bool) -> Vec<u8> {
    let mut xy = Vec::with_capacity(rows as usize * if multi { 4 } else { 2 });
    let mut ids = Vec::with_capacity(rows as usize);
    let mut offsets = Vec::with_capacity(rows as usize + 1);
    offsets.push(0);
    let mut valid = Vec::with_capacity(rows as usize);
    let mut starts = Vec::with_capacity(rows as usize);
    let mut ends = Vec::with_capacity(rows as usize);
    let mut sv = Vec::with_capacity(rows as usize);
    let mut ev = Vec::with_capacity(rows as usize);
    for i in first..first + rows {
        let x = if dense {
            (i % 100) as f64 * 1e-7
        } else {
            (mix(i) >> 11) as f64 / ((1u64 << 53) as f64) * 360. - 180.
        };
        let y = if dense {
            (i % 101) as f64 * 1e-7
        } else {
            (mix(i ^ 0xabcdef) >> 11) as f64 / ((1u64 << 53) as f64) * 160. - 80.
        };
        let visible = i % 1000 != 999;
        valid.push(u8::from(visible));
        if visible {
            xy.extend([x, y]);
            if multi {
                xy.extend([(x + 1e-7).min(180.), y]);
            }
        }
        offsets.push((xy.len() / 2) as u32);
        ids.push(if i % 23 == 0 { u64::MAX } else { u64::MAX - i });
        let a = (i % 1000) as i64 * 1000 - 500_000;
        starts.push(if i == 0 {
            i64::MIN
        } else if i == 1 {
            i64::MAX - 1
        } else {
            a
        });
        ends.push(if i == 0 {
            i64::MIN + 1
        } else if i == 1 {
            i64::MAX
        } else {
            a + 25_000
        });
        sv.push(u8::from(i % 7 != 0 || i < 2));
        ev.push(u8::from(i % 11 != 0 || i < 2));
    }
    let col = GeoColumn::from_descriptor(GeoDescriptor {
        geometry: if multi {
            GeoGeometry::MultiPoint
        } else {
            GeoGeometry::Point
        },
        crs: GeoCrs::Epsg4326,
        xy: &xy,
        validity: &valid,
        feature_ids: Some(&ids),
        offsets0: if multi { &offsets } else { &[] },
        offsets1: &[],
        offsets2: &[],
        limits: GeoLimits::default(),
    })
    .unwrap();
    GeoChunk::encode(
        &col,
        Some(GeoIntervals {
            starts: &starts,
            ends: &ends,
            start_validity: &sv,
            end_validity: &ev,
        }),
    )
    .unwrap()
}
fn source(rows: u64, multi: bool, dense: bool) -> (GeoSourceManifest, f64) {
    let started = Instant::now();
    let _credit = GeoProcessorLease::acquire(96 << 20).unwrap();
    let mut b = GeoManifestBuilder::new();
    for first in (0..rows).step_by(CHUNK as usize) {
        let bytes = chunk(first, (rows - first).min(CHUNK), multi, dense);
        b.push(&GeoChunk::parse(&bytes, 96 << 20).unwrap()).unwrap();
    }
    let m = b.finish(u64::MAX).unwrap();
    (m, started.elapsed().as_secs_f64() * 1000.)
}
struct Store {
    blob: File,
    offsets: File,
    pages: u64,
    bytes: u64,
    namespace: Option<u64>,
    write_cap: u64,
    root: PathBuf,
    next_free_check: u64,
}
impl Store {
    fn new(path: &Path, write_cap: u64) -> Self {
        assert!(fs::read_dir(path).unwrap().next().is_none());
        Self {
            blob: File::options()
                .create_new(true)
                .read(true)
                .write(true)
                .open(path.join("pages.bin"))
                .unwrap(),
            offsets: File::options()
                .create_new(true)
                .read(true)
                .write(true)
                .open(path.join("offsets.bin"))
                .unwrap(),
            pages: 0,
            bytes: 0,
            namespace: None,
            write_cap,
            root: path.to_path_buf(),
            next_free_check: 0,
        }
    }
    fn write(&mut self, t: &GeoHierarchyTicket, b: &[u8]) {
        if let Some(n) = self.namespace {
            assert_eq!(n, t.storage_namespace());
        } else {
            self.namespace = Some(t.storage_namespace());
        }
        assert_eq!(t.page_id(), self.pages + 1);
        assert_eq!(b.len(), t.encoded_bytes());
        if self.bytes >= self.next_free_check {
            let df = std::process::Command::new("df")
                .arg("-Pk")
                .arg(&self.root)
                .output()
                .unwrap();
            assert!(df.status.success());
            let text = String::from_utf8(df.stdout).unwrap();
            let available: u64 = text
                .lines()
                .last()
                .unwrap()
                .split_whitespace()
                .nth(3)
                .unwrap()
                .parse()
                .unwrap();
            assert!(
                available * 1024 >= (8u64 << 30) + (128 << 20),
                "free disk crossed explicit8GiB headroom; refusing another write"
            );
            self.next_free_check = self.bytes + (64 << 20);
        }
        let total = self.bytes.checked_add(b.len() as u64).unwrap();
        assert!(total <= self.write_cap);
        self.blob.seek(SeekFrom::Start(self.bytes)).unwrap();
        self.blob.write_all(b).unwrap();
        self.offsets.seek(SeekFrom::Start(self.pages * 16)).unwrap();
        self.offsets.write_all(&self.bytes.to_le_bytes()).unwrap();
        self.offsets
            .write_all(&(b.len() as u64).to_le_bytes())
            .unwrap();
        self.bytes = total;
        self.pages += 1;
    }
    fn read(&mut self, t: &GeoHierarchyTicket) -> Vec<u8> {
        assert_eq!(self.namespace, Some(t.storage_namespace()));
        assert!(t.page_id() > 0 && t.page_id() <= self.pages);
        self.offsets
            .seek(SeekFrom::Start((t.page_id() - 1) * 16))
            .unwrap();
        let mut index = [0; 16];
        self.offsets.read_exact(&mut index).unwrap();
        let offset = u64::from_le_bytes(index[..8].try_into().unwrap());
        let n = u64::from_le_bytes(index[8..].try_into().unwrap());
        assert_eq!(n, t.encoded_bytes() as u64);
        assert!(offset.checked_add(n).unwrap() <= self.bytes);
        self.blob.seek(SeekFrom::Start(offset)).unwrap();
        let mut bytes = vec![0; n as usize];
        self.blob.read_exact(&mut bytes).unwrap();
        bytes
    }
}
fn observe(peak: &mut usize) {
    *peak = (*peak).max(GeoProcessorLease::live_bytes());
}
fn build(
    m: &GeoSourceManifest,
    multi: bool,
    dense: bool,
    grid: u32,
    store: &mut Store,
    budget: QueryBudget,
    peak: &mut usize,
) -> (Arc<ValidatedGeoHierarchy>, u64, u64) {
    let mut s = GeoHierarchyBuildSession::new(
        m,
        GeoHierarchyOptions { grid },
        budget,
        m.rows() * if multi { 2 } else { 1 },
        store.write_cap,
    )
    .unwrap();
    let mut reads = 0;
    let mut source_bytes = 0;
    loop {
        observe(peak);
        match s.step().unwrap() {
            GeoHierarchyStep::NeedRead(t) => {
                observe(peak);
                let bytes = if let Some(r) = t.source_request() {
                    let bytes = chunk(r.first_row, r.rows as u64, multi, dense);
                    source_bytes += bytes.len() as u64;
                    bytes
                } else {
                    store.read(&t)
                };
                reads += bytes.len() as u64;
                s.supply(&t, &bytes).unwrap();
                drop(bytes);
                s.release_read(&t).unwrap();
            }
            GeoHierarchyStep::NeedWrite(t) => {
                observe(peak);
                store.write(&t, s.write_bytes(&t).unwrap());
                s.acknowledge_write(&t).unwrap();
            }
            GeoHierarchyStep::Complete => {
                assert_eq!(s.written_bytes(), store.bytes);
                return (s.finish().unwrap(), reads, source_bytes);
            }
            other => panic!("unexpected build {other:?}"),
        }
    }
}
fn query(
    index: Arc<ValidatedGeoHierarchy>,
    store: &mut Store,
    camera: GeoViewport,
    time: TimePredicate,
    options: GeoLodOptions,
    peak: &mut usize,
) -> (Option<GeoHierarchyResult>, u64, u64, &'static str) {
    let mut s = GeoHierarchyQuerySession::new(
        index,
        camera,
        time,
        options,
        u64::MAX,
        1,
        1,
        GeoHierarchyQueryLimits {
            leaf_reads: 1_000_000,
            directory_reads: 1_000_000,
            vertex_records: options.max_projected_vertices,
            read_bytes: 128 << 30,
            ..GeoHierarchyQueryLimits::default()
        },
    )
    .unwrap();
    loop {
        observe(peak);
        match s.step().unwrap() {
            GeoHierarchyStep::NeedRead(t) => {
                observe(peak);
                let bytes = store.read(&t);
                s.supply(&t, &bytes).unwrap();
                drop(bytes);
                s.release_read(&t).unwrap();
            }
            GeoHierarchyStep::Complete => return (Some(s.finish().unwrap()), 0, 0, "refined"),
            GeoHierarchyStep::FullScanFrontier => {
                return (
                    None,
                    s.stats().directory_reads,
                    s.stats().leaf_reads,
                    "frontier",
                )
            }
            GeoHierarchyStep::FullScanWork => {
                return (
                    None,
                    s.stats().directory_reads,
                    s.stats().leaf_reads,
                    "work",
                )
            }
            other => panic!("unexpected query {other:?}"),
        }
    }
}
fn equal(a: &GeoPointResult, b: &GeoPointResult) {
    assert_eq!(a.key, b.key);
    assert_eq!(a.visible_vertices, b.visible_vertices);
    assert_eq!(a.grid_capped, b.grid_capped);
    match (&a.output, &b.output) {
        (GeoPointOutput::Direct(a), GeoPointOutput::Direct(b)) => assert_eq!(a, b),
        (GeoPointOutput::Reduced(a), GeoPointOutput::Reduced(b)) => {
            assert_eq!(a.len(), b.len());
            for (a, b) in a.iter().zip(b) {
                assert_eq!(
                    (a.count, a.x.to_bits(), a.y.to_bits()),
                    (b.count, b.x.to_bits(), b.y.to_bits())
                );
            }
        }
        _ => panic!("tier mismatch"),
    }
}
fn percentile(values: &[f64], fraction: f64) -> f64 {
    let mut values = values.to_vec();
    values.sort_by(f64::total_cmp);
    values[((values.len() as f64 * fraction).ceil() as usize).saturating_sub(1)]
}
fn main() {
    let args: Vec<_> = std::env::args().collect();
    let rows: u64 = args[1].parse().unwrap();
    let grid: u32 = args[2].parse().unwrap();
    let multi = args[3] == "multipoint";
    let iterations: usize = args[4].parse().unwrap();
    let path = PathBuf::from(&args[5]);
    let cap: u64 = args[6].parse().unwrap();
    let dense = args.get(7).is_some_and(|s| s == "dense");
    let density = args.get(8).is_some_and(|s| s == "density");
    assert!(rows > 0 && rows <= 100_000_000 && (20..=100).contains(&iterations));
    assert!(["point", "multipoint"].contains(&args[3].as_str()));
    let mut store = Store::new(&path, cap);
    let budget = QueryBudget {
        max_rows_examined: rows * 2,
        max_read_bytes: 128 << 30,
        ..QueryBudget::default()
    };
    let (m, manifest_ms) = source(rows, multi, dense);
    let mut peak = 0;
    let started = Instant::now();
    let (index, reads, source_bytes) = build(&m, multi, dense, grid, &mut store, budget, &mut peak);
    let build_ms = started.elapsed().as_secs_f64() * 1000.;
    println!("{{\"phase\":\"cold\",\"rows\":{rows},\"vertices\":{},\"geometry\":{:?},\"grid\":{grid},\"manifest_ms\":{manifest_ms},\"build_ms\":{build_ms},\"source_read_bytes\":{source_bytes},\"all_read_bytes\":{reads},\"write_bytes\":{},\"offset_index_bytes\":{},\"pages\":{},\"root_ledger_bytes\":{},\"build_ledger_peak_bytes\":{peak},\"manifest_reservation_bytes\":100663296,\"write_cap_bytes\":{cap}}}",rows-rows/1000,args[3],store.bytes,store.pages*16,store.pages,index.reserved_bytes());
    let options = GeoLodOptions {
        kind: if density {
            geo_lod::GeoReducedKind::Density
        } else {
            geo_lod::GeoReducedKind::Cluster
        },
        max_projected_vertices: rows * 4,
        ..GeoLodOptions::default()
    };
    let times = [
        ("all", TimePredicate::All),
        ("instant", TimePredicate::Instant(0)),
        (
            "window",
            TimePredicate::Window {
                start: -5000,
                end: 5000,
            },
        ),
    ];
    for (name, lon, zoom) in [("zoom", 0., 8.), ("pan", 1., 8.), ("deep", 1., 12.)] {
        let camera =
            GeoViewport::new(GeoCrs::Epsg4326, lon, 0., zoom, 800., 600., 0., 0., true).unwrap();
        for (time_name, time) in times {
            let mut samples = Vec::with_capacity(iterations);
            let mut reference_ms = 0.;
            for trial in 0..iterations {
                let started = Instant::now();
                let (result, _, _, decision) =
                    query(index.clone(), &mut store, camera, time, options, &mut peak);
                let elapsed = started.elapsed().as_secs_f64() * 1000.;
                assert_eq!(decision, "refined");
                let result = result.unwrap();
                samples.push(elapsed);
                if trial == 0 {
                    let started = Instant::now();
                    let mut reference_credit = GeoProcessorLease::acquire(96 << 20).unwrap();
                    observe(&mut peak);
                    let reference = geo_lod::process(
                        &m,
                        &mut |r: ReadRequest| Ok(chunk(r.first_row, r.rows as u64, multi, dense)),
                        camera,
                        time,
                        u64::MAX,
                        1,
                        1,
                        options,
                        budget,
                        &mut || false,
                    )
                    .unwrap();
                    reference_ms = started.elapsed().as_secs_f64() * 1000.;
                    reference_credit
                        .resize(geo_lod::GeoPointLod::output_bytes(&reference))
                        .unwrap();
                    equal(&result.result, &reference);
                    let _scene_credit = GeoProcessorLease::acquire(64 << 20).unwrap();
                    observe(&mut peak);
                    let started = Instant::now();
                    let scene =
                        geo_lod_scene::compile(&result.result, GeoStyle::default(), 32 << 20)
                            .unwrap();
                    let scene_ms = started.elapsed().as_secs_f64() * 1000.;
                    let oracle =
                        geo_lod_scene::compile(&reference, GeoStyle::default(), 32 << 20).unwrap();
                    assert_eq!(scene.scene, oracle.scene);
                    let scene_path = store.root.join(format!("{name}-{time_name}.scene"));
                    fs::write(&scene_path, &scene.scene).unwrap();
                    println!("{{\"phase\":\"query-first\",\"name\":{name:?},\"time\":{time_name:?},\"query_ms\":{elapsed},\"scene_compile_ms\":{scene_ms},\"directory_reads\":{},\"leaf_reads\":{},\"read_bytes\":{},\"candidate_vertices\":{},\"visible_vertices\":{},\"passes\":{},\"selected_cells\":{},\"result_ledger_bytes\":{},\"scene_bytes\":{},\"scene_path\":{:?},\"reference_ms\":{reference_ms},\"parity\":\"exact-key-record-centroid-and-scene-bytes\"}}",result.stats.directory_reads,result.stats.leaf_reads,result.stats.bytes_read,result.stats.vertex_records,result.result.visible_vertices,result.stats.passes,result.stats.selected_cells,result.reserved_bytes(),scene.scene.len(),scene_path.to_string_lossy());
                }
            }
            println!("{{\"phase\":\"query-summary\",\"name\":{name:?},\"time\":{time_name:?},\"iterations\":{iterations},\"p50_ms\":{},\"p95_ms\":{},\"reference_ms\":{reference_ms},\"samples_ms\":{samples:?}}}",percentile(&samples,0.5),percentile(&samples,0.95));
        }
    }
    let camera = GeoViewport::new(GeoCrs::Epsg4326, 0., 0., 0., 800., 600., 0., 0., true).unwrap();
    let started = Instant::now();
    let (result, dirs, leaves, decision) = query(
        index.clone(),
        &mut store,
        camera,
        TimePredicate::All,
        options,
        &mut peak,
    );
    if dense {
        assert!(result.is_some());
    } else {
        assert!(result.is_none());
        assert_eq!(decision, "frontier");
        assert_eq!(leaves, 0);
    }
    drop(result);
    println!("{{\"phase\":\"broad-decision\",\"decision\":{decision:?},\"elapsed_ms\":{},\"directory_reads\":{dirs},\"leaf_reads\":{leaves},\"ledger_peak_bytes\":{peak}}}",started.elapsed().as_secs_f64()*1000.);
    drop(index);
    assert_eq!(GeoProcessorLease::live_bytes(), 0);
    println!("{{\"phase\":\"cleanup\",\"engine_ledger_bytes\":0}}");
}
