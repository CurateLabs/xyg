  executablePath:'/home/ubuntu/.cache/ms-playwright/chromium-1248/chrome-linux64/chrome',
  args:['--no-sandbox','--use-angle=swiftshader','--enable-unsafe-swiftshader','--ignore-gpu-blocklist'],
});
try {
  console.log(JSON.stringify(await qualifyStreamlitGraphForge(browser,
    'http://127.0.0.1:18617','/tmp/graphforge-streamlit-floor-evidence',paint),null,2));
} finally {await browser.close();}
